import { auth } from "./firebase.js"; import { createUserWithEmailAndPassword } from "https://www.gstatic.com/firebasejs/10.12.5/firebase-auth.js"; const registerBtn = document.getElementById("registerBtn"); registerBtn.addEventListener("click", () => { const email = document.getElementById("email").value.trim(); const password = document.getElementById("password").value; createUserWithEmailAndPassword(auth, email, password) .then((userCredential) => { alert("Registration Successful!"); window.location.href = "login.html"; }) .catch((error) => { alert(error.message); }); });
